static int maxprofit1(int[] arr)
	{
		int maxprofit=0;
		int dif=0;
		int buyday=0;
		int sellday=0;
		for(int i=0;i<arr.length;i++)
		{
			for(int j=i;j<arr.length;j++)
			{
				if(arr[i]<arr[j])
				{
					dif=arr[j]-arr[i];
					if(dif>maxprofit)
					{
						maxprofit=dif;
						buyday=i;
						sellday=j;
					}
				}
			}
		}
		System.out.println("buyday is:"+buyday);
		System.out.println("sellday is :"+sellday);
	
		return maxprofit;
		
	}
